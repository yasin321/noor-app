function getGuestNotifEnabled(): boolean {
  try {
    return localStorage.getItem('noor_guest_notif_enabled') === 'true';
  } catch {

  }

function AppContent() {
  const { user, profile, loading } = useAuth();
  const [tab, setTab] = useState<Tab>('home');
  const [navigationDirection, setNavigationDirection] = useState<'forward' | 'backward'>('forward');
  const [showAuth, setShowAuth] = useState(false);

  const tabOrder: Tab[] = ['home', 'tasbih', 'library', 'reminders', 'settings'];
  const navigateTo = (nextTab: Tab) => {
    if (nextTab === tab) return;
    setNavigationDirection(tabOrder.indexOf(nextTab) >= tabOrder.indexOf(tab) ? 'forward' : 'backward');
    setTab(nextTab);
  };

  const notifEnabled = profile?.notif_enabled ?? getGuestNotifEnabled();

  useEffect(() => {
    // Wait for the auth/profile state before rescheduling. Otherwise the initial
    // default (enabled) can briefly override a saved disabled preference.
    if (!loading && Capacitor.isNativePlatform()) {
      void rescheduleAll(notifEnabled);
    }
  }, [loading, user?.id, notifEnabled]);

  if (loading) {
    return (
      <div className="min-h-screen flex items-center justify-center" style={{ background: 'var(--bg)' }}>
        <div className="w-14 h-14 rounded-full animate-pulse" style={{ background: 'var(--primary)', opacity: 0.3 }} />
      </div>
    );
  }

  if (showAuth && !user) {
    return <AuthScreen onBack={() => setShowAuth(false)} />;
  }

  return (
    <div className="min-h-screen" style={{ background: 'var(--bg)', color: 'var(--text)' }}>
      <main key={tab} className={`page-transition page-transition-${navigationDirection}`} aria-live="polite">
        {tab === 'home' && <HomeScreen onNavigate={navigateTo} />}
        {tab === 'tasbih' && <TasbihCounter />}
        {tab === 'library' && <LibraryScreen onNavigate={navigateTo} />}
        {tab === 'reminders' && <RemindersScreen />}
        {tab === 'settings' && <SettingsScreen onExitGuest={() => setShowAuth(true)} />}
      </main>
      <BottomNav active={tab} onChange={navigateTo} />
    </div>
  );
}

        <AppContent />
      </ThemeProvider>

const ROLLING_HORIZON_DAYS = 14;

const DAY_NAMES_FA = ['شنبه', 'یکشنبه', 'دوشنبه', 'سه‌شنبه', 'چهارشنبه', 'پنجشنبه', 'جمعه'];

function ourDayToJsDay(ourDay: number): number {

function jsDayToOurDay(jsDay: number): number {
  return (jsDay + 1) % 7;

export function getNextOccurrence(
  time: string,
  days: number[],
  fromNow: Date = new Date(),
  horizonDays: number = ROLLING_HORIZON_DAYS,
): Date | null {
  const [h, m] = time.split(':').map(Number);
  if (isNaN(h) || isNaN(m)) return null;

  for (let i = 0; i <= horizonDays; i++) {
    const candidate = new Date(fromNow.getFullYear(), fromNow.getMonth(), fromNow.getDate() + i, h, m, 0, 0);
    const jsDay = candidate.getDay();
    const ourDay = jsDayToOurDay(jsDay);
    const dayMatches = days.length === 0 || days.includes(ourDay);
    if (dayMatches && candidate.getTime() > fromNow.getTime()) {
      return candidate;
    }
  }
  return null;
}

export function getUpcomingOccurrences(
  time: string,
  days: number[],
  fromNow: Date = new Date(),
  horizonDays: number = ROLLING_HORIZON_DAYS,
): Date[] {
  const result: Date[] = [];
  const [h, m] = time.split(':').map(Number);
  if (isNaN(h) || isNaN(m)) return result;

  for (let i = 0; i <= horizonDays; i++) {
    const candidate = new Date(fromNow.getFullYear(), fromNow.getMonth(), fromNow.getDate() + i, h, m, 0, 0);
    const jsDay = candidate.getDay();
    const ourDay = jsDayToOurDay(jsDay);
    const dayMatches = days.length === 0 || days.includes(ourDay);
    if (dayMatches && candidate.getTime() > fromNow.getTime()) {
      result.push(candidate);
    }
  }
  return result;
}

export function parseDateTime(dateStr: string, time: string): Date | null {
  const [h, m] = time.split(':').map(Number);
  if (isNaN(h) || isNaN(m)) return null;
  const [y, mo, d] = dateStr.split('-').map(Number);
  if (isNaN(y) || isNaN(mo) || isNaN(d)) return null;
  return new Date(y, mo - 1, d, h, m, 0, 0);
}

export function isOneTimePast(date: string | null, time: string, now: Date = new Date()): boolean {
  if (!date) return true;
  const dt = parseDateTime(date, time);
  if (!dt) return true;
  return dt.getTime() <= now.getTime();
}

function reminderOccurrenceId(reminderId: string, occurrenceIndex: number): number {
  let hash = 0;
  const str = `${reminderId}:${occurrenceIndex}`;
  for (let i = 0; i < str.length; i++) {
    hash = ((hash << 5) - hash) + str.charCodeAt(i);
    hash = hash & 0x7fffffff;
  }
  return (hash % 900000) + 100000;
}

export function getReminderNotificationIds(reminderId: string, count: number = ROLLING_HORIZON_DAYS + 2): number[] {
  const ids: number[] = [];
  for (let i = 0; i < count; i++) {
    ids.push(reminderOccurrenceId(reminderId, i));
  }
  return ids;
}

export async function cancelReminder(reminderId: string): Promise<ScheduleResult> {

export async function scheduleReminder(reminder: LocalReminder, notifEnabled: boolean = true): Promise<ScheduleResult> {
  if (!reminder.enabled) {
    await cancelReminder(reminder.id);
    return { success: true, scheduledIds: [] };
  }

  if (!notifEnabled) {
    await cancelReminder(reminder.id);
    return { success: true, scheduledIds: [] };
  }

  const perm = await checkPermission();
  if (perm !== 'granted') {
    return { success: false, error: 'اجازه اعلان داده نشده است' };
  }

  // Exact-alarm access is an Android-only setting. Calling this API on web/iOS
  // can fail even though those platforms do not require Android's exact-alarm grant.
  if (Capacitor.getPlatform() === 'android') {
    try {
      const { exact_alarm } = await LocalNotifications.checkExactNotificationSetting();
      if (exact_alarm !== 'granted') {
        return {
          success: false,
          error: 'اجازه آلارم‌ها و یادآوری‌ها داده نشده است؛ دسترسی آلارم دقیق اندروید را بررسی کن.',
        };
      }
    } catch {
      return {
        success: false,
        error: 'وضعیت اجازه آلارم دقیق قابل بررسی نیست؛ تنظیمات اعلان اندروید را بررسی کن.',
      };
    }
  }

  await ensureChannel();

  const dhikr = reminder.dhikrId ? getDhikr(reminder.dhikrId) : null;
  const body = reminder.body || (dhikr ? dhikr.translation : 'یادآوری ذکر');

  await cancelReminder(reminder.id);

  if (reminder.type === 'once') {
    if (isOneTimePast(reminder.date, reminder.time)) {
      return { success: false, error: 'تاریخ و زمان یادآوری یکباره در گذشته است' };
    }
    const at = parseDateTime(reminder.date ?? '', reminder.time);
    if (!at) {
      return { success: false, error: 'تاریخ یا زمان نامعتبر است' };
    }
    const notifId = reminderOccurrenceId(reminder.id, 0);
    try {
      await LocalNotifications.schedule({
        notifications: [{
          id: notifId,
          title: reminder.title,
          body,
          channelId: 'noor-reminders',
          schedule: { at, allowWhileIdle: true },
          smallIcon: 'ic_stat_icon',
        }],
      });
      return { success: true, scheduledIds: [notifId] };
    } catch (err) {
      return { success: false, error: `خطا در زمان‌بندی: ${err}` };
    }
  }

  const occurrences = getUpcomingOccurrences(reminder.time, reminder.days);
  if (occurrences.length === 0) {
    return { success: false, error: 'هیچ زمان مناسبی برای زمان‌بندی یافت نشد' };
  }

  const scheduledIds: number[] = [];
  const failed: string[] = [];

  for (let i = 0; i < occurrences.length; i++) {
    const notifId = reminderOccurrenceId(reminder.id, i);
    try {
      await LocalNotifications.schedule({
        notifications: [{
          id: notifId,
          title: reminder.title,
          body,
          channelId: 'noor-reminders',
          schedule: { at: occurrences[i], allowWhileIdle: true },
          smallIcon: 'ic_stat_icon',
        }],
      });
      scheduledIds.push(notifId);
    } catch (err) {
      failed.push(`${occurrences[i].toISOString()}: ${err}`);
    }
  }

  if (scheduledIds.length === 0) {
    return { success: false, error: `خطا در زمان‌بندی همه موارد: ${failed.join('; ')}` };
  }
  return { success: true, scheduledIds };
}

export async function createReminder(params: {
  title: string;
  time: string;
  days: number[];
  dhikrId: string | null;
  body?: string;
  type: 'recurring' | 'once';
  date?: string | null;
  notifEnabled?: boolean;
}): Promise<{ reminder: LocalReminder; schedule: ScheduleResult }> {
  const id = generateReminderId();
  const now = new Date().toISOString();

  if (params.type === 'once') {
    if (isOneTimePast(params.date ?? null, params.time)) {
      return {
        reminder: {
          id, title: params.title, body: params.body ?? '', time: params.time,
          days: [], enabled: false, dhikrId: params.dhikrId,
          type: 'once', date: params.date ?? null,
          createdAt: now, updatedAt: now,
        },
        schedule: { success: false, error: 'تاریخ و زمان در گذشته است' },
      };
    }
  }

  const reminder: LocalReminder = {
    id,
    title: params.title,
    body: params.body ?? '',
    time: params.time,
    days: params.type === 'once' ? [] : params.days,
    enabled: true,
    dhikrId: params.dhikrId,
    type: params.type,
    date: params.type === 'once' ? (params.date ?? null) : null,
    createdAt: now,
    updatedAt: now,
  };

  saveReminder(reminder);
  const notifEnabled = params.notifEnabled ?? true;
  const schedule = await scheduleReminder(reminder, notifEnabled);

  if (!schedule.success) {
    reminder.enabled = false;
    reminder.updatedAt = new Date().toISOString();
    saveReminder(reminder);
  }

  return { reminder, schedule };
}

export async function updateReminder(
  id: string,
  updates: Partial<Pick<LocalReminder, 'title' | 'time' | 'days' | 'dhikrId' | 'body' | 'enabled' | 'type' | 'date'>>,
  notifEnabled: boolean = true,
): Promise<{ reminder: LocalReminder | null; schedule: ScheduleResult }> {
  const all = loadReminders();
  const existing = all.find((r) => r.id === id);
  if (!existing) return { reminder: null, schedule: { success: false, error: 'یادآوری یافت نشد' } };

  const updated: LocalReminder = {
    ...existing,
    ...updates,
    updatedAt: new Date().toISOString(),
  };
  saveReminder(updated);

  await cancelReminder(id);
  const schedule = await scheduleReminder(updated, notifEnabled);

  if (!schedule.success && updates.enabled !== false) {
    updated.enabled = false;
    updated.updatedAt = new Date().toISOString();
    saveReminder(updated);
  }

  return { reminder: updated, schedule };
}

export async function toggleReminder(id: string, notifEnabled: boolean = true): Promise<{ reminder: LocalReminder | null; schedule: ScheduleResult }> {
  const all = loadReminders();
  const existing = all.find((r) => r.id === id);
  if (!existing) return { reminder: null, schedule: { success: false, error: 'یادآوری یافت نشد' } };

  if (existing.enabled) {
    await cancelReminder(id);
    return updateReminder(id, { enabled: false }, notifEnabled);
  } else {
    return updateReminder(id, { enabled: true }, notifEnabled);
  }
}

export async function deleteReminder(id: string): Promise<ScheduleResult> {
  const schedule = await cancelReminder(id);
  deleteReminderFromStorage(id);
  return schedule;
}

export async function rescheduleAll(notifEnabled: boolean = true): Promise<{ total: number; succeeded: number; failed: number }> {
  const all = loadReminders();
  let succeeded = 0;
  let failed = 0;

  for (const reminder of all) {
    if (!reminder.enabled) continue;
    const result = await scheduleReminder(reminder, notifEnabled);
    if (result.success) succeeded++;
    else failed++;
  }

  return { total: all.filter((reminder) => reminder.enabled).length, succeeded, failed };
}

export async function getPendingNotifications(): Promise<number[]> {
  try {
    const { notifications } = await LocalNotifications.getPending();
    return notifications.map((n) => n.id);
  } catch {
    return [];
  }
}

export async function cancelAllScheduledNotifications(): Promise<void> {
  const all = loadReminders();
  for (const r of all) {
    await cancelReminder(r.id);
  }
}

export async function rescheduleAllReminders(
  notifEnabled: boolean = true
): Promise<{ total: number; succeeded: number; failed: number }> {
  return rescheduleAll(notifEnabled);
}

type ReminderStore = {
  version: number;
  reminders: Record<string, LocalReminder>;
};

const STORAGE_KEY = 'noor_reminders_store';
const SCHEMA_VERSION = 2;

function defaultStore(): ReminderStore {
  return { version: SCHEMA_VERSION, reminders: {} };
}

function migrate(raw: unknown): ReminderStore {
  if (!raw || typeof raw !== 'object') return defaultStore();
  const obj = raw as Partial<ReminderStore> & { reminders?: Record<string, Record<string, unknown>> };

  if (!obj.version || obj.version < 1) return defaultStore();
  if (!obj.reminders || typeof obj.reminders !== 'object') return defaultStore();

  const reminders: Record<string, LocalReminder> = {};
  for (const [id, r] of Object.entries(obj.reminders)) {
    if (!r || typeof r.title !== 'string' || typeof r.time !== 'string' ||
        !Array.isArray(r.days) || typeof r.enabled !== 'boolean') {
      continue;
    }
    const type = (r.type === 'once' || r.type === 'recurring')
      ? (r.type as ReminderType)
      : 'recurring';
    const date = typeof r.date === 'string' ? r.date : null;
    reminders[id] = {
      id,
      title: r.title,
      body: typeof r.body === 'string' ? r.body : '',
      time: r.time,
      days: r.days as number[],
      enabled: r.enabled,
      dhikrId: typeof r.dhikrId === 'string' ? r.dhikrId : null,
      type,
      date,
      createdAt: typeof r.createdAt === 'string' ? r.createdAt : new Date().toISOString(),
      updatedAt: typeof r.updatedAt === 'string' ? r.updatedAt : new Date().toISOString(),
    };
  }
  return { version: SCHEMA_VERSION, reminders };
}

export function loadReminders(): LocalReminder[] {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    if (!raw) return [];
    const store = migrate(JSON.parse(raw));
    return Object.values(store.reminders).sort((a, b) => a.time.localeCompare(b.time));
  } catch {
    return [];
  }
}

export function loadReminder(id: string): LocalReminder | null {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    if (!raw) return null;
    const store = migrate(JSON.parse(raw));
    return store.reminders[id] ?? null;
  } catch {
    return null;
  }
}

function saveReminders(reminders: Record<string, LocalReminder>): void {
  try {
    const store: ReminderStore = { version: SCHEMA_VERSION, reminders };
    localStorage.setItem(STORAGE_KEY, JSON.stringify(store));
  } catch {
    // storage full or unavailable
  }
}

export function saveReminder(reminder: LocalReminder): void {
  const all = loadReminders();
  const map: Record<string, LocalReminder> = {};
  for (const r of all) map[r.id] = r;
  map[reminder.id] = reminder;
  saveReminders(map);
}

export function deleteReminderFromStorage(id: string): void {
  const all = loadReminders();
  const map: Record<string, LocalReminder> = {};
  for (const r of all) if (r.id !== id) map[r.id] = r;
  saveReminders(map);
}

export function clearAllReminders(): void {
  try {
    localStorage.removeItem(STORAGE_KEY);
  } catch {
    // ignore
  }
}

export function generateReminderId(): string {
  return `r_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`;
}

      if (result.failed > 0) {
        setNotifMessage(
          `از ${result.total} یادآوری ذخیره‌شده، ${result.succeeded} مورد زمان‌بندی شد و ${result.failed} مورد ناموفق بود. تنظیمات اعلان و آلارم دقیق گوشی را بررسی کن.`
        );
      } else {
        setNotifMessage(
          `${result.succeeded} یادآوری دوباره زمان‌بندی شد. توجه: این تنظیم به‌تنهایی پخش فایل صوتی اذان را تضمین نمی‌کند.`
        );
      }
    } else {
      await cancelAllScheduledNotifications();
      setNotifMessage('اعلان‌های یادآوری خاموش شدند.');
    }
  };

  const requestNotificationAccess = async () => {
    setNotifMessage(null);
    let permission = await checkPermission();
    if (permission !== 'granted') permission = await requestPermission();

    if (permission !== 'granted') {
      setNotifMessage('اندروید اجازهٔ اعلان را نداد. از تنظیمات گوشی ← برنامه‌ها ← نور ← اعلان‌ها آن را فعال کن.');
      return;
    }

    if (Capacitor.getPlatform() === 'android') {
      try {
        let { exact_alarm } = await LocalNotifications.checkExactNotificationSetting();
        if (exact_alarm !== 'granted') {
          const result = await LocalNotifications.changeExactNotificationSetting();
          exact_alarm = result.exact_alarm;
        }

        if (exact_alarm !== 'granted') {
          setNotifMessage('دسترسی آلارم دقیق هنوز فعال نیست. آن را در صفحهٔ تنظیمات اندروید فعال کن و دوباره بررسی را بزن.');
          return;
        }
      } catch {
        setNotifMessage('نتوانستم صفحهٔ دسترسی آلارم دقیق را باز یا بررسی کنم. از تنظیمات گوشی ← برنامه‌ها ← نور ← آلارم‌ها و یادآوری‌ها بررسی کن.');
        return;
      }
    }

    if (profile) {
      await updateProfile({ notif_enabled: true });
    } else {
      setGuestNotifEnabled(true);
      setGuestNotif(true);
    }

    const result = await rescheduleAllReminders(true);

    if (result.failed > 0) {
      setNotifMessage(
        `اجازه‌ها بررسی شد؛ ${result.succeeded} یادآوری زمان‌بندی شد و ${result.failed} مورد ناموفق بود. تنظیمات اعلان و آلارم دقیق اندروید را بررسی کن.`
      );
    } else {
      setNotifMessage(`اجازه‌ها بررسی شد؛ ${result.succeeded} یادآوری زمان‌بندی شد.`);
    }
  };

  const currentCityId = profile?.city_id ?? guestCityId;
  const currentCity = getCityById(currentCityId);

  const applyCity = async (id: string) => {
    if (profile) {
      await updateProfile({ city_id: id });
    } else {
      setGuestCityId(id);
      setGuestCityIdState(id);
    }
  };

  const applyTheme = async (id: string) => {
    setThemeId(id);
    if (profile) {
      await updateProfile({ theme_id: id });
    }
  };

  const applyAdhan = async (id: string) => {
    if (profile) {
      await updateProfile({ adhan_id: id });
    } else {
      try { localStorage.setItem(GUEST_ADHAN_KEY, id); } catch { /* ignore unavailable storage */ }
      setGuestAdhanId(id);
    }
  };

  const applyMethod = async (id: string) => {
    if (profile) {
      await updateProfile({ calculation_method: id });
    }
  };

  const currentMethod = profile?.calculation_method ?? 'jafari';
  const methods = getAvailableMethods();

  if (section === 'theme') {
    return (
      <div className="px-5 pt-6 pb-28 animate-fade-in">
        <div className="flex items-center gap-3 mb-5">
          <button onClick={() => setSection('main')} className="text-muted text-sm">بازگشت</button>
          <h1 className="text-2xl font-bold text-app">پوسته‌ها</h1>
        </div>
        <div className="grid grid-cols-2 gap-3">
          {themes.map((t) => (
            <button
              key={t.id}
              onClick={() => applyTheme(t.id)}
              className="card-flat p-4 text-right relative overflow-hidden"
              style={{ borderColor: themeId === t.id ? t.primary : theme.border }}
            >
              <div className="w-full h-20 rounded-xl mb-3" style={{ background: t.gradient }} />
              <p className="text-app font-bold text-sm">{t.nameFa}</p>
              <p className="text-muted text-xs">{t.name}</p>
              {themeId === t.id && (
                <div className="absolute top-2 left-2 w-6 h-6 rounded-full flex items-center justify-center" style={{ background: t.primary }}>
                  <Check size={14} color={t.bg} />
                </div>
              )}
            </button>
          ))}
        </div>
      </div>
    );
  }

  if (section === 'city') {
    const grouped = CITIES.reduce<Record<string, City[]>>((acc, c) => {
      (acc[c.countryFa] ??= []).push(c);
      return acc;
    }, {});

    return (
      <div className="px-5 pt-6 pb-28 animate-fade-in">
        <div className="flex items-center gap-3 mb-5">
          <button onClick={() => setSection('main')} className="text-muted text-sm">بازگشت</button>
          <h1 className="text-2xl font-bold text-app">انتخاب شهر</h1>
        </div>
        <p className="text-muted text-xs mb-4">شهر خود را برای محاسبه اوقات شرعی انتخاب کنید</p>
        {Object.entries(grouped).map(([country, cities]) => (
          <div key={country} className="mb-5">
            <p className="text-muted text-xs font-semibold mb-2">{country}</p>
            <div className="space-y-2">
              {cities.map((c) => (
                <button
                  key={c.id}
                  onClick={() => applyCity(c.id)}
                  className="w-full card-flat p-4 flex items-center justify-between"
                  style={{ borderColor: currentCityId === c.id ? theme.primary : theme.border }}
                >
                  <div>
                    <p className="text-app font-medium text-sm">{c.nameFa}</p>
                    <p className="text-muted text-xs" dir="ltr">{c.timezone}</p>
                  </div>
                  {currentCityId === c.id && (
                    <Check size={18} color={theme.primary} />
                  )}
                </button>
              ))}
            </div>
          </div>
        ))}
      </div>
    );
  }

  if (section === 'method') {
    return (
      <div className="px-5 pt-6 pb-28 animate-fade-in">
        <div className="flex items-center gap-3 mb-5">
          <button onClick={() => setSection('main')} className="text-muted text-sm">بازگشت</button>
          <h1 className="text-2xl font-bold text-app">روش محاسبه</h1>
        </div>
        <p className="text-muted text-xs mb-4">روش محاسبه اوقات شرعی را انتخاب کنید</p>
        <div className="space-y-2">
          {methods.map((m) => (
            <button
              key={m.id}
              onClick={() => applyMethod(m.id)}
              className="w-full card-flat p-4 flex items-center justify-between"
              style={{ borderColor: currentMethod === m.id ? theme.primary : theme.border }}
            >
              <p className="text-app font-medium text-sm">{m.name}</p>
              {currentMethod === m.id && (
                <Check size={18} color={theme.primary} />
              )}
            </button>
          ))}
        </div>
        {!profile && (
          <p className="text-muted text-xs mt-4 opacity-60">
            برای ذخیره روش محاسبه باید وارد حساب شوید. در حالت مهمان، روش جعفری استفاده می‌شود.
          </p>
        )}
      </div>
    );
  }

  if (section === 'adhan') {
    return (
      <div className="px-5 pt-6 pb-28 animate-fade-in">
        <div className="flex items-center gap-3 mb-5">
          <button onClick={() => setSection('main')} className="text-muted text-sm">بازگشت</button>
          <h1 className="text-2xl font-bold text-app">اذان‌ها</h1>
        </div>
        <p className="text-muted text-xs mb-3">انتخاب صدا و شنیدن نمونهٔ اذان‌های ایرانی</p>
        <p className="text-muted text-xs mb-4 leading-relaxed">توجه: در این مرحله فقط نمونهٔ ماهور آنلاین پخش می‌شود؛ پخش خودکار اذان در وقت نماز و هنگام قفل بودن گوشی هنوز پیاده‌سازی نشده است.</p>
        <div className="space-y-3">
          {adhans.map((a) => (
            <div
              key={a.id}
              className="w-full card-flat p-4"
              style={{ borderColor: (profile?.adhan_id ?? guestAdhanId) === a.id ? theme.primary : theme.border }}
            >
              <button
                type="button"
                onClick={() => a.audioUrl && applyAdhan(a.id)}
                disabled={!a.audioUrl}
                aria-label={a.audioUrl ? `انتخاب ${a.nameFa}` : `${a.nameFa}، فایل صوتی هنوز آماده نیست`}
                className="w-full flex items-center justify-between text-right disabled:cursor-not-allowed disabled:opacity-60"
              >
                <div>
                  <p className="text-app font-medium text-sm">{a.nameFa}</p>
                  <p className="text-muted text-xs mt-1">{a.reciter}</p>
                  {!a.audioUrl && <p className="text-muted text-[11px] mt-1">به‌زودی؛ فایل صوتی هنوز آماده نیست</p>}
                </div>
                {(profile?.adhan_id ?? guestAdhanId) === a.id && (
                  <Check size={18} color={theme.primary} />
                )}
              </button>
              {a.audioUrl ? (
                <div className="mt-3">
                  <audio controls preload="none" src={a.audioUrl} className="w-full" aria-label={`نمونهٔ صوتی ${a.nameFa}`} />
                  <p className="text-muted text-xs mt-2 leading-relaxed">
                    مجوز: {a.license}. صاحب اثر: {a.reciter}.
                    {' '}<a href={a.sourceUrl} target="_blank" rel="noreferrer" className="underline">منبع فایل</a>
                    {' '}· <a href={a.licenseUrl} target="_blank" rel="noreferrer" className="underline">متن مجوز</a>
                  </p>
                </div>
              ) : (
                <p className="text-muted text-xs mt-2">فایل صوتی این گزینه هنوز به‌صورت مجاز به برنامه اضافه نشده است.</p>
              )}
            </div>
          ))}
        </div>
      </div>
    );
  }

  return (
    <div className="px-5 pt-6 pb-28 animate-fade-in">
      <h1 className="text-2xl font-bold text-app mb-6">تنظیمات</h1>

      <div className="card p-5 mb-4 flex items-center gap-4">
        <div className="w-14 h-14 rounded-2xl flex items-center justify-center" style={{ background: theme.primary }}>
          <User size={24} color={theme.bg} />
        </div>
        <div className="flex-1">
          <p className="text-app font-bold">{profile?.display_name || (user ? 'کاربر' : 'مهمان')}</p>
          <p className="text-muted text-xs" dir="ltr">{user?.email || 'حالت آفلاین'}</p>
        </div>
      </div>

      <div className="space-y-2">
        <button onClick={() => setSection('theme')} className="w-full card-flat p-4 flex items-center justify-between">
          <div className="flex items-center gap-3">
            <Palette size={20} color={theme.primary} />
            <span className="text-app font-medium text-sm">پوسته و رنگ</span>
          </div>
          <span className="text-muted text-xs">{themes.find((t) => t.id === themeId)?.nameFa}</span>
        </button>

        <button onClick={() => setSection('city')} className="w-full card-flat p-4 flex items-center justify-between">
          <div className="flex items-center gap-3">
            <MapPin size={20} color={theme.primary} />
            <span className="text-app font-medium text-sm">شهر و موقعیت</span>
          </div>
          <span className="text-muted text-xs">{currentCity.nameFa}</span>
        </button>

        <button onClick={() => setSection('method')} className="w-full card-flat p-4 flex items-center justify-between">
          <div className="flex items-center gap-3">
            <MapPin size={20} color={theme.primary} />
            <span className="text-app font-medium text-sm">روش محاسبه</span>
          </div>
          <span className="text-muted text-xs">{methods.find((m) => m.id === currentMethod)?.name}</span>
        </button>

        <button onClick={() => setSection('adhan')} className="w-full card-flat p-4 flex items-center justify-between">
          <div className="flex items-center gap-3">
            <Volume2 size={20} color={theme.primary} />
            <span className="text-app font-medium text-sm">صدای اذان</span>
          </div>
          <span className="text-muted text-xs">{adhans.find((a) => a.id === (profile?.adhan_id ?? guestAdhanId))?.nameFa}</span>
        </button>

        <div className="card-flat p-4">
          <div className="flex items-center justify-between">
            <div className="flex items-center gap-3">
              <Bell size={20} color={theme.primary} />
              <span className="text-app font-medium text-sm">اعلان‌های یادآوری</span>
            </div>
            <button
              onClick={toggleNotifications}
              className="w-11 h-6 rounded-full transition-all relative"
              style={{ background: notifEnabled ? theme.primary : `color-mix(in srgb, ${theme.textMuted} 25%, transparent)` }}
            >
              <div className="absolute top-0.5 w-5 h-5 rounded-full bg-white transition-all" style={{ right: notifEnabled ? '2px' : '22px' }} />
            </button>
          </div>
          {!notifSupported && (
            <div className="flex items-start gap-2 mt-3 pt-3" style={{ borderTop: `1px solid ${theme.border}` }}>
              <AlertCircle size={14} color={theme.accent} className="flex-shrink-0 mt-0.5" />
              <p className="text-muted text-xs leading-relaxed">
                اعلان زمان‌بندی‌شده فقط در نسخه اندروید کار می‌کند. در مرورگر، یادآوری‌ها فقط وقتی برنامه باز است نمایش داده می‌شوند.
              </p>
            </div>
          )}
          {notifMessage && (
            <div className="mt-3 p-3 rounded-xl" style={{ background: `color-mix(in srgb, ${theme.accent} 10%, transparent)`, border: `1px solid ${theme.border}` }}>
              <p className="text-muted text-xs leading-relaxed">{notifMessage}</p>
              {notifSupported && (
                <button type="button" onClick={requestNotificationAccess} className="mt-2 text-sm font-bold" style={{ color: theme.primary }}>
                  بررسی / درخواست اجازهٔ اعلان
                </button>
              )}
            </div>
          )}
        </div>
      </div>

      {user ? (
        <button onClick={signOut} className="w-full mt-6 py-3 rounded-2xl flex items-center justify-center gap-2 text-sm font-medium" style={{ background: 'rgba(239,68,68,0.12)', color: '#fca5a5' }}>
          <LogOut size={18} /> خروج از حساب
        </button>
      ) : onExitGuest ? (
        <button onClick={onExitGuest} className="w-full mt-6 py-3 rounded-2xl flex items-center justify-center gap-2 text-sm font-bold" style={{ background: theme.primary, color: theme.bg }}>
          <LogIn size={18} /> ورود به حساب
        </button>
      ) : null}
    </div>
  );
}
