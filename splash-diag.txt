package com.noor.app;

import android.content.BroadcastReceiver;
import android.content.Context;
import android.content.Intent;
import android.os.Build;
import android.os.PowerManager;

public class AdhanAlarmReceiver extends BroadcastReceiver {
    public static final String ACTION_TRIGGER = "com.noor.app.ACTION_TRIGGER_ADHAN_ALARM";
    public static final String ACTION_STOP = "com.noor.app.ACTION_STOP_ADHAN_ALARM";

    @Override
    public void onReceive(Context context, Intent intent) {
        if (intent == null) return;
        String action = intent.getAction();

        // ۱. اگر رویداد ریستارت گوشی، آپدیت برنامه یا تغییر ساعت بود، تمام آلارم‌های آینده را در AlarmManager بازنشانی کن
        if (Intent.ACTION_BOOT_COMPLETED.equals(action) ||
                Intent.ACTION_LOCKED_BOOT_COMPLETED.equals(action) ||
                "android.intent.action.QUICKBOOT_POWERON".equals(action) ||
                Intent.ACTION_MY_PACKAGE_REPLACED.equals(action) ||
                "android.intent.action.TIME_SET".equals(action) ||
                Intent.ACTION_TIMEZONE_CHANGED.equals(action)) {
            NoorAlarmPlugin.restorePersistedAlarms(context);
            return;
        }

        // ۲. اگر دستور توقف آلارم بود
        if (ACTION_STOP.equals(action) || AdhanForegroundService.ACTION_STOP.equals(action)) {
            AdhanForegroundService.killActivePlayer(context);
            Intent stopService = new Intent(context, AdhanForegroundService.class);
            stopService.setAction(AdhanForegroundService.ACTION_STOP);
            context.stopService(stopService);
            context.sendBroadcast(new Intent(AdhanAlarmActivity.ACTION_CLOSE_UI));
            return;
        }

        // ۳. اجرای آلارم اذان، پیش از اذان یا تعقیبات (حتی اگر action خالی باشد ولی دارای title یا id باشد)
        if (ACTION_TRIGGER.equals(action) || intent.hasExtra("title") || intent.hasExtra("id")) {
            // گرفتن WakeLock فوری ۱۰ ثانیه‌ای برای تضمین بالا آمدن سرویس روی صفحه قفل خاموش
            try {
                PowerManager pm = (PowerManager) context.getSystemService(Context.POWER_SERVICE);
                if (pm != null) {
                    PowerManager.WakeLock wl = pm.newWakeLock(
                            PowerManager.PARTIAL_WAKE_LOCK | PowerManager.ACQUIRE_CAUSES_WAKEUP,
                            "noor:ReceiverTransitionWakeLock"
                    );
                    wl.acquire(10000L);
                }
            } catch (Exception ignored) {}

            int alarmId = intent.getIntExtra("id", -1);
            if (alarmId != -1) {
                NoorAlarmPlugin.removePersistedAlarm(context, alarmId);
            }

            Intent serviceIntent = new Intent(context, AdhanForegroundService.class);
            serviceIntent.putExtra("title", intent.getStringExtra("title"));
            serviceIntent.putExtra("body", intent.getStringExtra("body"));
            serviceIntent.putExtra("soundType", intent.getStringExtra("soundType"));
            serviceIntent.putExtra("adhanId", intent.getStringExtra("adhanId"));
            serviceIntent.putExtra("durationSeconds", intent.getIntExtra("durationSeconds", 0));
            serviceIntent.putExtra("volumePercent", intent.getIntExtra("volumePercent", 100));

            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
                context.startForegroundService(serviceIntent);
            } else {
                context.startService(serviceIntent);
            }
        }
    }
}

                    }
                } catch (Exception ignored) {}
                try {
                    activePlayer.reset();
                    activePlayer.release();
                } catch (Exception ignored) {}
                activePlayer = null;
            }
            if (ctx != null) {
                try {
                    Vibrator vib = (Vibrator) ctx.getSystemService(Context.VIBRATOR_SERVICE);
                    if (vib != null) vib.cancel();
                } catch (Exception ignored) {}
                try {
                    NotificationManager nm = (NotificationManager) ctx.getSystemService(Context.NOTIFICATION_SERVICE);
                    if (nm != null) nm.cancel(NOTIF_ID);

    public static void scheduleExactWithSystem(
            Context ctx,
            int alarmId,
            long targetTime,
            String title,
            String body,
            String soundType,
            String adhanId,
            int durationSeconds,
            int volumePercent
    ) {
        AlarmManager am = (AlarmManager) ctx.getSystemService(Context.ALARM_SERVICE);
        if (am == null) return;

        Intent intent = new Intent(ctx, AdhanAlarmReceiver.class);
        intent.setAction(AdhanAlarmReceiver.ACTION_TRIGGER);
        intent.setData(Uri.parse("noor://alarm/" + alarmId));

        intent.putExtra("title", title);
        intent.putExtra("body", body);
        intent.putExtra("soundType", soundType);
        intent.putExtra("adhanId", adhanId);
        intent.putExtra("durationSeconds", durationSeconds);
        intent.putExtra("volumePercent", volumePercent);

        PendingIntent pi = PendingIntent.getBroadcast(
                ctx,

                intent,
                PendingIntent.FLAG_UPDATE_CURRENT | PendingIntent.FLAG_IMMUTABLE
        );

            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.LOLLIPOP) {
                Intent showIntent = new Intent(ctx, MainActivity.class);
                PendingIntent showPi = PendingIntent.getActivity(
                        ctx,
                        alarmId,

                        PendingIntent.FLAG_UPDATE_CURRENT | PendingIntent.FLAG_IMMUTABLE
                );
                AlarmManager.AlarmClockInfo clockInfo = new AlarmManager.AlarmClockInfo(targetTime, showPi);
                am.setAlarmClock(clockInfo, pi);
            } else if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M) {
                am.setExactAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, targetTime, pi);
            } else {
                am.setExact(AlarmManager.RTC_WAKEUP, targetTime, pi);
            }
        } catch (SecurityException se) {
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M) {
                am.setAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, targetTime, pi);
            } else {
                am.set(AlarmManager.RTC_WAKEUP, targetTime, pi);
            }
        }

    // بازیابی خودکار تمام آلارم‌های آینده از حافظه دائم نیتیو بلافاصله پس از روشن شدن یا ریستارت گوشی
    public static void restorePersistedAlarms(Context ctx) {
        try {
            SharedPreferences prefs = ctx.getSharedPreferences(ALARM_STORE_PREFS, Context.MODE_PRIVATE);
            Map<String, ?> all = prefs.getAll();
            if (all == null || all.isEmpty()) return;
            long now = System.currentTimeMillis();
            SharedPreferences.Editor editor = prefs.edit();
            for (Map.Entry<String, ?> entry : all.entrySet()) {
                try {
                    String rawJson = String.valueOf(entry.getValue());
                    JSONObject obj = new JSONObject(rawJson);
                    int id = obj.optInt("id", -1);
                    long triggerAt = obj.optLong("triggerAtMillis", 0L);
                    if (id == -1 || triggerAt <= now) {
                        editor.remove(entry.getKey());
                        continue;
                    }
                    scheduleExactWithSystem(
                            ctx,
                            id,
                            triggerAt,
                            obj.optString("title", "وقت اذان و نیایش"),
                            obj.optString("body", "حَیَّ عَلَی الصَّلَاة"),
                            obj.optString("soundType", "adhan"),
                            obj.optString("adhanId", "mahur_saeed"),
                            obj.optInt("durationSeconds", 0),
                            obj.optInt("volumePercent", 100)
                    );
                } catch (Exception ignored) {}
            }
            editor.apply();
        } catch (Exception ignored) {}
    }

    private void fetchFusedHighAccuracyLocation(final PluginCall call) {
        final Context ctx = getContext();
        boolean fine = ContextCompat.checkSelfPermission(ctx, Manifest.permission.ACCESS_FINE_LOCATION) == PackageManager.PERMISSION_GRANTED;
        boolean coarse = ContextCompat.checkSelfPermission(ctx, Manifest.permission.ACCESS_COARSE_LOCATION) == PackageManager.PERMISSION_GRANTED;
        if (!fine && !coarse) {
            JSObject ret = new JSObject();
            ret.put("hasCoords", false);
            call.resolve(ret);
            return;
        }

        try {
            final FusedLocationProviderClient fusedClient = LocationServices.getFusedLocationProviderClient(ctx);
            final boolean[] resolved = new boolean[]{false};

            CurrentLocationRequest currentReq = new CurrentLocationRequest.Builder()
                    .setPriority(Priority.PRIORITY_HIGH_ACCURACY)
                    .setGranularity(Granularity.GRANULARITY_FINE)
                    .setMaxUpdateAgeMillis(0)
                    .setDurationMillis(4500)
                    .build();

            CancellationTokenSource cts = new CancellationTokenSource();
            fusedClient.getCurrentLocation(currentReq, cts.getToken())
                    .addOnSuccessListener(loc -> {
                        if (!resolved[0] && loc != null) {
                            resolved[0] = true;
                            resolveWithLocation(call, ctx, loc, "gms_fused_current");
                        }
                    });

            LocationRequest liveReq = new LocationRequest.Builder(Priority.PRIORITY_HIGH_ACCURACY, 500)
                    .setMinUpdateIntervalMillis(250)
                    .setWaitForAccurateLocation(false)
                    .setMaxUpdates(3)
                    .build();

            final LocationCallback callback = new LocationCallback() {
                @Override
                public void onLocationResult(@NonNull LocationResult result) {
                    Location best = result.getLastLocation();
                    if (best == null) return;
                    if (!resolved[0] && (!best.hasAccuracy() || best.getAccuracy() <= 65f)) {
                        resolved[0] = true;
                        fusedClient.removeLocationUpdates(this);
                        resolveWithLocation(call, ctx, best, "gms_fused_stream");
                    }
                }
            };

            fusedClient.requestLocationUpdates(liveReq, callback, Looper.getMainLooper());

            new android.os.Handler(Looper.getMainLooper()).postDelayed(() -> {
                if (!resolved[0]) {
                    try { fusedClient.removeLocationUpdates(callback); } catch (Exception ignored) {}
                    fusedClient.getLastLocation().addOnSuccessListener(lastLoc -> {
                        if (!resolved[0] && lastLoc != null) {
                            resolved[0] = true;
                            resolveWithLocation(call, ctx, lastLoc, "gms_fused_last");
                        } else if (!resolved[0]) {
                            fallbackToAndroidGpsManager(call, ctx);
                        }
                    }).addOnFailureListener(e -> {
                        if (!resolved[0]) fallbackToAndroidGpsManager(call, ctx);
                    });
                }
            }, 3800L);

        } catch (Exception e) {
            fallbackToAndroidGpsManager(call, ctx);
        }
    }

    private void fallbackToAndroidGpsManager(PluginCall call, Context ctx) {
        try {
            LocationManager lm = (LocationManager) ctx.getSystemService(Context.LOCATION_SERVICE);
            if (lm != null) {
                Location gps = lm.isProviderEnabled(LocationManager.GPS_PROVIDER) ? lm.getLastKnownLocation(LocationManager.GPS_PROVIDER) : null;
                if (gps != null) {
                    resolveWithLocation(call, ctx, gps, "android_gps");
                    return;
                }
            }
        } catch (Exception ignored) {}
        JSObject ret = new JSObject();
        ret.put("hasCoords", false);
        call.resolve(ret);
    }

    private boolean downloadFileUrl(String urlStr, File dest) {
        try {
            URL url = new URL(urlStr);
            HttpURLConnection conn = (HttpURLConnection) url.openConnection();
            conn.setConnectTimeout(10000);
            conn.setReadTimeout(15000);
            conn.connect();
            if (conn.getResponseCode() != HttpURLConnection.HTTP_OK) return false;
            InputStream is = conn.getInputStream();
            FileOutputStream fos = new FileOutputStream(dest);
            byte[] buf = new byte[8192];
            int len;
            while ((len = is.read(buf)) > 0) {
                fos.write(buf, 0, len);
            }
            fos.flush();
            fos.close();
            is.close();
            return dest.length() > 1024;
        } catch (Exception e) {
            return false;
        }
    }

            if (files != null) {

                    if (f.length() > 1024) {
                        String name = f.getName().replace(".mp3", "").replace(".ogg", "");
                        arr.put(name);
                    }
                }
            }
        }
        JSObject ret = new JSObject();
        ret.put("ids", arr);
        call.resolve(ret);
    }

    @PluginMethod
    public void previewAdhanAudio(PluginCall call) {
        String adhanId = call.getString("adhanId", "mahur_saeed");
        Integer durationSeconds = call.getInt("durationSeconds", 15);
        Integer volumePercent = call.getInt("volumePercent", 95);
        Context ctx = getContext();
        Intent srv = new Intent(ctx, AdhanForegroundService.class);
        srv.putExtra("title", "پیش‌نمایش صوتی نور");
        srv.putExtra("body", "در حال پخش نمونه صوتی...");
        srv.putExtra("soundType", "adhan");
        srv.putExtra("adhanId", adhanId);
        srv.putExtra("durationSeconds", durationSeconds != null ? durationSeconds : 15);
        srv.putExtra("volumePercent", volumePercent != null ? volumePercent : 95);
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            ctx.startForegroundService(srv);
        } else {
            ctx.startService(srv);
        }
        call.resolve();
    }

    @PluginMethod
    public void stopPreviewAudio(PluginCall call) {
        Context ctx = getContext();
        AdhanForegroundService.killActivePlayer(ctx);
        Intent stopIntent = new Intent(ctx, AdhanForegroundService.class);
        stopIntent.setAction(AdhanForegroundService.ACTION_STOP);
        ctx.startService(stopIntent);
        call.resolve();
    }

public class NoorPrayerWidgetProvider extends AppWidgetProvider {
    private static final String PREFS_NAME = "noor_widget_prefs";

    // تولید هوشمند ذکر متناسب با روز هفته و ساعت شبانه‌روز به صورت خودکار
    public static String getDynamicHourlyAndDailyDhikr() {
        Calendar cal = Calendar.getInstance();
        int dayOfWeek = cal.get(Calendar.DAY_OF_WEEK);
        int hour = cal.get(Calendar.HOUR_OF_DAY);

        String dayDhikr;
        String dayName;
        switch (dayOfWeek) {
            case Calendar.SATURDAY:
                dayName = "شنبه";
                dayDhikr = "يَا رَبَّ الْعَالَمِينَ";
                break;
            case Calendar.SUNDAY:
                dayName = "یکشنبه";
                dayDhikr = "يَا ذَا الْجَلَالِ وَالْإِكْرَامِ";
                break;
            case Calendar.MONDAY:
                dayName = "دوشنبه";
                dayDhikr = "يَا قَاضِيَ الْحَاجَاتِ";
                break;
            case Calendar.TUESDAY:
                dayName = "سه‌شنبه";
                dayDhikr = "يَا أَرْحَمَ الرَّاحِمِينَ";
                break;
            case Calendar.WEDNESDAY:
                dayName = "چهارشنبه";
                dayDhikr = "يَا حَيُّ يَا قَيُّومُ";
                break;
            case Calendar.THURSDAY:
                dayName = "پنجشنبه";
                dayDhikr = "لَا إِلَهَ إِلَّا اللَّهُ الْمَلِكُ الْحَقُّ الْمُبِينُ";
                break;
            case Calendar.FRIDAY:
            default:
                dayName = "جمعه";
                dayDhikr = "اللَّهُمَّ صَلِّ عَلَى مُحَمَّدٍ وَآلِ مُحَمَّدٍ وَعَجِّلْ فَرَجَهُمْ";
                break;
        }

        String timeMoment;
        if (hour >= 3 && hour < 6) {
            timeMoment = "🌌 ذکر سحر و صبح: اللَّهُمَّ بَارِكْ لِمَوْلَانَا صَاحِبِ الزَّمَانِ";
        } else if (hour >= 6 && hour < 11) {
            timeMoment = "☀️ ذکر آغاز روز (" + dayName + "): " + dayDhikr + " (۱۰۰ مرتبه)";
        } else if (hour >= 11 && hour < 15) {
            timeMoment = "🕌 وقت فضیلت ظهر (" + dayName + "): " + dayDhikr;
        } else if (hour >= 15 && hour < 18) {
            timeMoment = "🌿 ذکر عصرگاهی: أَسْتَغْفِرُ اللَّهَ رَبِّي وَأَتُوبُ إِلَيْهِ";
        } else if (hour >= 18 && hour < 21) {
            timeMoment = "🌙 ذکر شامگاه (" + dayName + "): " + dayDhikr;
        } else {
            timeMoment = "✨ پیش از خواب: سوره مبارکه واقعه و تسبیحات حضرت زهرا (س)";
        }
        return timeMoment;
    }

    public static void pushWidgetUpdate(Context context, String title, String nextPrayer, String subtitle) {
        try {
            SharedPreferences prefs = context.getSharedPreferences(PREFS_NAME, Context.MODE_PRIVATE);
            if (title != null && nextPrayer != null && subtitle != null) {
                prefs.edit()
                    .putString("title", title)
                    .putString("nextPrayer", nextPrayer)
                    .putString("subtitle", subtitle)
                    .apply();
            }
            String savedTitle = prefs.getString("title", "📍 به افق شهر شما");
            String savedNext = prefs.getString("nextPrayer", "اذان بعدی: در حال محاسبه");
            String savedSub = prefs.getString("subtitle", "همراه معنوی نور");
            String fajr = prefs.getString("fajr", "۰۴:۵۵");
            String sunrise = prefs.getString("sunrise", "۰۶:۲۰");
            String dhuhr = prefs.getString("dhuhr", "۱۱:۵۰");
            String maghrib = prefs.getString("maghrib", "۱۷:۲۵");
            String midnight = prefs.getString("midnight", "۲۳:۰۵");

            AppWidgetManager gm = AppWidgetManager.getInstance(context);
            ComponentName cn = new ComponentName(context, NoorPrayerWidgetProvider.class);
            int[] ids = gm.getAppWidgetIds(cn);
            if (ids != null && ids.length > 0) {
                Intent intent = new Intent(context, MainActivity.class);
                PendingIntent pendingIntent = PendingIntent.getActivity(
                    context, 0, intent, PendingIntent.FLAG_UPDATE_CURRENT | PendingIntent.FLAG_IMMUTABLE
                );
                String dynamicDhikr = getDynamicHourlyAndDailyDhikr();
                for (int appWidgetId : ids) {
                    RemoteViews views = new RemoteViews(context.getPackageName(), R.layout.noor_prayer_widget);
                    views.setTextViewText(R.id.widget_title, savedTitle);
                    views.setTextViewText(R.id.widget_next_prayer, savedNext);
                    views.setTextViewText(R.id.widget_subtitle, savedSub);
                    views.setTextViewText(R.id.widget_time_fajr, fajr);
                    views.setTextViewText(R.id.widget_time_sunrise, sunrise);
                    views.setTextViewText(R.id.widget_time_dhuhr, dhuhr);
                    views.setTextViewText(R.id.widget_time_maghrib, maghrib);
                    views.setTextViewText(R.id.widget_time_midnight, midnight);
                    views.setTextViewText(R.id.widget_dynamic_dhikr, dynamicDhikr);
                    views.setOnClickPendingIntent(R.id.widget_root, pendingIntent);
                    gm.updateAppWidget(appWidgetId, views);
                }
            }
        } catch (Exception ignored) {
        }
    }

    @Override
    public void onUpdate(Context context, AppWidgetManager appWidgetManager, int[] appWidgetIds) {
        pushWidgetUpdate(context, null, null, null);
    }
