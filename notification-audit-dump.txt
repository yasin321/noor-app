    @PluginMethod
    public void scheduleNativeAlarm(PluginCall call) {
        Integer id = call.getInt("id", 1001);
        Long triggerAtMillis = call.getLong("triggerAtMillis", System.currentTimeMillis() + 5000L);
        String title = call.getString("title", "وقت اذان و نیایش");
        String body = call.getString("body", "حَیَّ عَلَی الصَّلَاة");
        String soundType = call.getString("soundType", "adhan");
        String adhanId = call.getString("adhanId", "mahur_saeed");
        Integer durationSeconds = call.getInt("durationSeconds", 0);
        Integer volumePercent = call.getInt("volumePercent", 100);

        Context ctx = getContext();
        AlarmManager am = (AlarmManager) ctx.getSystemService(Context.ALARM_SERVICE);
        Intent intent = new Intent(ctx, AdhanAlarmReceiver.class);
        intent.putExtra("id", id != null ? id : 1001);
        intent.putExtra("title", title);
        intent.putExtra("body", body);
        intent.putExtra("soundType", soundType);
        intent.putExtra("adhanId", adhanId);
        intent.putExtra("durationSeconds", durationSeconds != null ? durationSeconds : 0);
        intent.putExtra("volumePercent", volumePercent != null ? volumePercent : 100);

        PendingIntent pi = PendingIntent.getBroadcast(
                ctx,
                id != null ? id : 1001,
                intent,
                PendingIntent.FLAG_UPDATE_CURRENT | PendingIntent.FLAG_IMMUTABLE
        );

        long targetTime = triggerAtMillis != null ? triggerAtMillis : System.currentTimeMillis() + 5000L;
        if (am != null) {
            try {
                if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.LOLLIPOP) {
                    Intent showIntent = new Intent(ctx, MainActivity.class);
                    PendingIntent showPi = PendingIntent.getActivity(
                            ctx, id != null ? id : 1001, showIntent,
                            PendingIntent.FLAG_UPDATE_CURRENT | PendingIntent.FLAG_IMMUTABLE
                    );
                    AlarmManager.AlarmClockInfo clockInfo = new AlarmManager.AlarmClockInfo(targetTime, showPi);
                    am.setAlarmClock(clockInfo, pi);
                } else if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M) {
                    am.setExactAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, targetTime, pi);
                } else {
                    am.setExact(AlarmManager.RTC_WAKEUP, targetTime, pi);
                }
            } catch (SecurityException se) {
                if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M) {
                    am.setAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, targetTime, pi);
                } else {
                    am.set(AlarmManager.RTC_WAKEUP, targetTime, pi);
                }
            }
        }
        call.resolve();
    }

    // دریافت آدرس آفلاین از Geocoder داخلی سیستم‌عامل اندروید (برای مواقع بدون اینترنت یا مکمل نشان)
    private String getOfflineAndroidAddress(Context ctx, double lat, double lng) {
        try {
            Geocoder geocoder = new Geocoder(ctx, new Locale("fa", "IR"));
            List<Address> addresses = geocoder.getFromLocation(lat, lng, 1);
            if (addresses != null && !addresses.isEmpty()) {
                Address a = addresses.get(0);
                StringBuilder sb = new StringBuilder();
                if (a.getLocality() != null) sb.append(a.getLocality());
                if (a.getSubLocality() != null) {
                    if (sb.length() > 0) sb.append("، ");
                    sb.append(a.getSubLocality());
                }
                if (a.getThoroughfare() != null) {
                    if (sb.length() > 0) sb.append("، ");
                    sb.append(a.getThoroughfare());
                }
                if (sb.length() > 0) return sb.toString();
                if (a.getAddressLine(0) != null) return a.getAddressLine(0);
            }
        } catch (Exception ignored) {}
        return "";
    }

    private void resolveWithLocation(PluginCall call, Context ctx, Location loc, String providerName) {
        JSObject ret = new JSObject();
        ret.put("hasCoords", true);
        ret.put("lat", loc.getLatitude());
        ret.put("lng", loc.getLongitude());
        ret.put("accuracy", loc.hasAccuracy() ? Math.round(loc.getAccuracy()) : 12);
        ret.put("provider", providerName);
        String nativeAddr = getOfflineAndroidAddress(ctx, loc.getLatitude(), loc.getLongitude());
        if (nativeAddr != null && !nativeAddr.isEmpty()) {
            ret.put("nativeAddress", nativeAddr);
        }
        call.resolve(ret);
    }

    // پیاده‌سازی دقیق مستندات رسمی نشان با Google Play Services FusedLocationProviderClient
    private void fetchFusedHighAccuracyLocation(final PluginCall call) {
        final Context ctx = getContext();
        boolean fine = ContextCompat.checkSelfPermission(ctx, Manifest.permission.ACCESS_FINE_LOCATION) == PackageManager.PERMISSION_GRANTED;
        boolean coarse = ContextCompat.checkSelfPermission(ctx, Manifest.permission.ACCESS_COARSE_LOCATION) == PackageManager.PERMISSION_GRANTED;
        if (!fine && !coarse) {
            JSObject ret = new JSObject();
            ret.put("hasCoords", false);
            call.resolve(ret);
            return;
        }

        try {
            final FusedLocationProviderClient fusedClient = LocationServices.getFusedLocationProviderClient(ctx);
            final boolean[] resolved = new boolean[]{false};

            // ۱. درخواست موقعیت لحظه‌ای (Fresh High-Accuracy Fix) با maxUpdateAgeMillis = 0 (بدون کش قدیمی!)
            CurrentLocationRequest currentReq = new CurrentLocationRequest.Builder()
                    .setPriority(Priority.PRIORITY_HIGH_ACCURACY)
                    .setGranularity(Granularity.GRANULARITY_FINE)
                    .setMaxUpdateAgeMillis(0)
                    .setDurationMillis(4500)
                    .build();

            CancellationTokenSource cts = new CancellationTokenSource();
            fusedClient.getCurrentLocation(currentReq, cts.getToken())
                    .addOnSuccessListener(loc -> {
                        if (!resolved[0] && loc != null) {
                            resolved[0] = true;
                            resolveWithLocation(call, ctx, loc, "gms_fused_current");
                        }
                    });

            // ۲. اجرای هم‌زمان requestLocationUpdates با بازه ۵۰۰ میلی‌ثانیه برای قفل فوری مثل اپ نشان
            LocationRequest liveReq = new LocationRequest.Builder(Priority.PRIORITY_HIGH_ACCURACY, 500)
                    .setMinUpdateIntervalMillis(250)
                    .setWaitForAccurateLocation(false)
                    .setMaxUpdates(3)
                    .build();

            final LocationCallback callback = new LocationCallback() {
                @Override
                public void onLocationResult(@NonNull LocationResult result) {
                    Location best = result.getLastLocation();
                    if (best == null) return;
                    // اگر دقت عالی (زیر ۶۵ متر) است، بلافاصله قفل کن
                    if (!resolved[0] && (!best.hasAccuracy() || best.getAccuracy() <= 65f)) {
                        resolved[0] = true;
                        fusedClient.removeLocationUpdates(this);
                        resolveWithLocation(call, ctx, best, "gms_fused_stream");
                    }
                }
            };

            fusedClient.requestLocationUpdates(liveReq, callback, Looper.getMainLooper());

            // ۳. تایم‌اوت ایمنی ۳.۸ ثانیه در صورت بودن در زیرزمین یا گوشی بدون گوگل‌پلی
            new android.os.Handler(Looper.getMainLooper()).postDelayed(() -> {
                if (!resolved[0]) {
                    try { fusedClient.removeLocationUpdates(callback); } catch (Exception ignored) {}
                    fusedClient.getLastLocation().addOnSuccessListener(lastLoc -> {
                        if (!resolved[0] && lastLoc != null) {
                            resolved[0] = true;
                            resolveWithLocation(call, ctx, lastLoc, "gms_fused_last");
                        } else if (!resolved[0]) {
                            fallbackToAndroidGpsManager(call, ctx);
                        }
                    }).addOnFailureListener(e -> {
                        if (!resolved[0]) fallbackToAndroidGpsManager(call, ctx);
                    });
                }
            }, 3800L);

        } catch (Exception e) {
            fallbackToAndroidGpsManager(call, ctx);
        }
    }

    private void fallbackToAndroidGpsManager(PluginCall call, Context ctx) {
        try {
            LocationManager lm = (LocationManager) ctx.getSystemService(Context.LOCATION_SERVICE);
            if (lm != null) {
                Location gps = lm.isProviderEnabled(LocationManager.GPS_PROVIDER) ? lm.getLastKnownLocation(LocationManager.GPS_PROVIDER) : null;
                if (gps != null) {
                    resolveWithLocation(call, ctx, gps, "android_gps");
                    return;
                }
            }
        } catch (Exception ignored) {}
        JSObject ret = new JSObject();
        ret.put("hasCoords", false);
        call.resolve(ret);
    }

    @PluginMethod
    public void downloadAdhanAudio(PluginCall call) {
        final String adhanId = call.getString("adhanId", "mahur_saeed");
        final String audioUrl = call.getString("audioUrl", "");
        final String fallbackUrl = call.getString("fallbackUrl", "");
        if (audioUrl == null || audioUrl.isEmpty()) {
            JSObject ret = new JSObject();
            ret.put("downloaded", false);
            call.resolve(ret);
            return;
        }
        new Thread(() -> {
            File dir = new File(getContext().getFilesDir(), "adhans");
            if (!dir.exists()) dir.mkdirs();
            String ext = audioUrl.endsWith(".ogg") ? ".ogg" : ".mp3";
            File outFile = new File(dir, adhanId + ext);
            boolean ok = downloadFileUrl(audioUrl, outFile);
            if (!ok && fallbackUrl != null && !fallbackUrl.isEmpty()) {
                ok = downloadFileUrl(fallbackUrl, outFile);
            }
            JSObject ret = new JSObject();
            ret.put("downloaded", ok);
            if (ok) ret.put("path", outFile.getAbsolutePath());
            call.resolve(ret);
        }).start();
    }

    private boolean downloadFileUrl(String urlStr, File dest) {
        try {
            URL url = new URL(urlStr);
            HttpURLConnection conn = (HttpURLConnection) url.openConnection();
            conn.setConnectTimeout(10000);
            conn.setReadTimeout(15000);
            conn.connect();
            if (conn.getResponseCode() != HttpURLConnection.HTTP_OK) return false;
            InputStream is = conn.getInputStream();
            FileOutputStream fos = new FileOutputStream(dest);
            byte[] buf = new byte[8192];
            int len;
            while ((len = is.read(buf)) > 0) {
                fos.write(buf, 0, len);
            }
            fos.flush();
            fos.close();
            is.close();
            return dest.length() > 1024;
        } catch (Exception e) {
            return false;
        }
    }

package com.noor.app;

import android.content.BroadcastReceiver;
import android.content.Context;
import android.content.Intent;
import android.os.Build;

public class AdhanAlarmReceiver extends BroadcastReceiver {
    public static final String ACTION_TRIGGER = "com.noor.app.ACTION_TRIGGER_ADHAN_ALARM";
    public static final String ACTION_STOP = "com.noor.app.ACTION_STOP_ADHAN_ALARM";

    @Override
    public void onReceive(Context context, Intent intent) {
        if (intent == null || intent.getAction() == null) return;

        if (ACTION_STOP.equals(intent.getAction())) {
            Intent stopService = new Intent(context, AdhanForegroundService.class);
            stopService.setAction(ACTION_STOP);
            context.stopService(stopService);
            context.sendBroadcast(new Intent("com.noor.app.ACTION_CLOSE_ALARM_UI"));
            return;
        }

        if (ACTION_TRIGGER.equals(intent.getAction())) {
            Intent serviceIntent = new Intent(context, AdhanForegroundService.class);
            serviceIntent.putExtra("title", intent.getStringExtra("title"));
            serviceIntent.putExtra("body", intent.getStringExtra("body"));
            serviceIntent.putExtra("soundType", intent.getStringExtra("soundType"));
            serviceIntent.putExtra("adhanId", intent.getStringExtra("adhanId"));
            serviceIntent.putExtra("durationSeconds", intent.getIntExtra("durationSeconds", 0));
            serviceIntent.putExtra("volumePercent", intent.getIntExtra("volumePercent", 90));

            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
                context.startForegroundService(serviceIntent);
            } else {
                context.startService(serviceIntent);
            }
        }
    }
}

public class AdhanForegroundService extends Service {
    public static final String CHANNEL_ID = "noor_fullscreen_alarm_v6";
    public static final String ACTION_STOP = "com.noor.app.ACTION_STOP_ADHAN";
    private static final int NOTIF_ID = 88001;

    private static final Object PLAYER_LOCK = new Object();
    private static MediaPlayer activePlayer = null;
    private static Handler stopHandler = new Handler(Looper.getMainLooper());
    private static Runnable stopRunnable = null;
    private PowerManager.WakeLock wakeLock;

    public static void killActivePlayer(Context ctx) {
        synchronized (PLAYER_LOCK) {
            if (stopRunnable != null) {
                stopHandler.removeCallbacks(stopRunnable);
                stopRunnable = null;
            }
            if (activePlayer != null) {
                try {
                    if (activePlayer.isPlaying()) {
                        activePlayer.stop();
                    }
                } catch (Exception ignored) {}
                try {
                    activePlayer.reset();
                    activePlayer.release();
                } catch (Exception ignored) {}
                activePlayer = null;
            }
            if (ctx != null) {
                try {
                    Vibrator vib = (Vibrator) ctx.getSystemService(Context.VIBRATOR_SERVICE);
                    if (vib != null) vib.cancel();
                } catch (Exception ignored) {}
                try {
                    NotificationManager nm = (NotificationManager) ctx.getSystemService(Context.NOTIFICATION_SERVICE);
                    if (nm != null) nm.cancelAll();
                } catch (Exception ignored) {}
            }
        }
    }

    @Override
    public void onCreate() {
        super.onCreate();
        createAlarmChannel();
    }

    @Override
    public int onStartCommand(Intent intent, int flags, int startId) {
        if (intent != null && ACTION_STOP.equals(intent.getAction())) {
            killActivePlayer(this);
            sendBroadcast(new Intent(AdhanAlarmActivity.ACTION_CLOSE_UI));
            stopForeground(true);
            stopSelf();
            return START_NOT_STICKY;
        }

        // همیشه قبل از شروع صدای جدید، هر صدای در حال پخش قبلی را ۱۰۰٪ قطع کن تا تداخل رخ ندهد
        killActivePlayer(this);
        acquireWakeLock();

        String title = intent != null && intent.getStringExtra("title") != null
                ? intent.getStringExtra("title")

    private void startAudioPlayback(String soundType, String adhanId, int durationSeconds, int volumePercent) {
        if ("silent".equals(soundType) || volumePercent <= 0) {
            return;
        }
        synchronized (PLAYER_LOCK) {
            try {
                MediaPlayer mp = new MediaPlayer();
                activePlayer = mp;
                mp.setAudioAttributes(
                        new AudioAttributes.Builder()
                                .setUsage(AudioAttributes.USAGE_ALARM)
                                .setContentType(AudioAttributes.CONTENT_TYPE_MUSIC)
                                .build()
                );

                boolean loaded = false;
                File downloadedDir = new File(getFilesDir(), "adhans");
                File customMp3 = new File(downloadedDir, adhanId + ".mp3");
                File customOgg = new File(downloadedDir, adhanId + ".ogg");

                if (customMp3.exists() && customMp3.length() > 1024) {
                    mp.setDataSource(customMp3.getAbsolutePath());
                    loaded = true;
                } else if (customOgg.exists() && customOgg.length() > 1024) {
                    mp.setDataSource(customOgg.getAbsolutePath());
                    loaded = true;
                }

                if (!loaded) {
                    String[] candidates = new String[] {
                            "public/adhans/" + adhanId + ".mp3",
                            "public/adhans/" + adhanId + ".ogg",
                            "public/adhans/chant_salawat.mp3",
                            "public/adhans/mahur_saeed.ogg"
                    };
                    for (String path : candidates) {
                        try {
                            AssetFileDescriptor afd = getAssets().openFd(path);
                            mp.setDataSource(afd.getFileDescriptor(), afd.getStartOffset(), afd.getLength());
                            afd.close();
                            loaded = true;
                            break;
                        } catch (Exception ignored) {}
                    }
                }

                if (!loaded) {
                    int rawId = getResources().getIdentifier("adhan_mahur", "raw", getPackageName());
                    if (rawId != 0) {
                        AssetFileDescriptor afd = getResources().openRawResourceFd(rawId);
                        mp.setDataSource(afd.getFileDescriptor(), afd.getStartOffset(), afd.getLength());
                        afd.close();
                        loaded = true;
                    }
                }

                if (!loaded) {
                    Uri fallbackUri = RingtoneManager.getDefaultUri(RingtoneManager.TYPE_ALARM);
                    mp.setDataSource(this, fallbackUri);
                }

                float vol = Math.max(0.05f, Math.min(1.0f, volumePercent / 100.0f));
                mp.setVolume(vol, vol);
                mp.setLooping(false);
                mp.setOnCompletionListener(player -> {
                    killActivePlayer(AdhanForegroundService.this);
                    sendBroadcast(new Intent(AdhanAlarmActivity.ACTION_CLOSE_UI));
                    stopForeground(true);
                    stopSelf();
                });
                mp.prepare();
                mp.start();

                int maxSeconds = durationSeconds > 0 ? durationSeconds : ("short_10s".equals(soundType) ? 16 : 210);
                stopRunnable = () -> {
                    killActivePlayer(AdhanForegroundService.this);
                    sendBroadcast(new Intent(AdhanAlarmActivity.ACTION_CLOSE_UI));
                    stopForeground(true);
                    stopSelf();
                };
                stopHandler.postDelayed(stopRunnable, maxSeconds * 1000L);
            } catch (Exception e) {
                killActivePlayer(this);
            }
        }
    }

    private void triggerShortVibration() {
        try {
            Vibrator v = (Vibrator) getSystemService(Context.VIBRATOR_SERVICE);
            if (v != null && v.hasVibrator()) {
                if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
                    v.vibrate(VibrationEffect.createWaveform(new long[]{0, 400, 250, 400}, -1));
                } else {
                    v.vibrate(600);
                }
            }
        } catch (Exception ignored) {}
    }

    private void acquireWakeLock() {
        try {
            PowerManager pm = (PowerManager) getSystemService(Context.POWER_SERVICE);
            if (pm != null) {
                wakeLock = pm.newWakeLock(
                        PowerManager.PARTIAL_WAKE_LOCK | PowerManager.ACQUIRE_CAUSES_WAKEUP,
                        "noor:AdhanAlarmWakeLock"
                );
                wakeLock.acquire(4 * 60 * 1000L);
            }
        } catch (Exception ignored) {}
    }

    private void createAlarmChannel() {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            NotificationChannel channel = new NotificationChannel(
                    CHANNEL_ID,
                    "پخش تمام‌صفحه اذان و تواشیح نور",

            );

            channel.setBypassDnd(true);
            channel.setLockscreenVisibility(Notification.VISIBILITY_PUBLIC);
            NotificationManager nm = getSystemService(NotificationManager.class);
            if (nm != null) nm.createNotificationChannel(channel);
        }
    }

    @Override
    public void onDestroy() {
        killActivePlayer(this);
        try {
            if (wakeLock != null && wakeLock.isHeld()) wakeLock.release();
        } catch (Exception ignored) {}
        super.onDestroy();
    }

    @Override
    public IBinder onBind(Intent intent) {
        return null;
    }
}

function resolvePrayerAlarmParams(reminderId: string, reminderTitle = ''): {

      soundType: 'alarm',

      volumePercent: 100,
    };

  return {
    soundType: globalMode === '10s' ? 'short_10s' : 'adhan',
    adhanId: selectedAdhan,
    durationSeconds: globalDurSec,
    volumePercent: 100,
  };
}

function computeDynamicPrayerDate(reminder: LocalReminder, baseCandidate: Date): Date {
  try {
    const isAutoPrayer = reminder.id.startsWith('prayer_auto_');
    const isPreAdhan =
      reminder.id.startsWith('pre_adhan_') ||
      reminder.title.includes('دقیقه تا اذان') ||
      reminder.title.includes('مانده تا اذان');
    if (!isAutoPrayer && !isPreAdhan) return baseCandidate;

    const city = getCityById(getGuestCityId());
    const exactGps = getCustomGpsCoords();
    const lat = exactGps ? exactGps.lat : city.lat;
    const lng = exactGps ? exactGps.lng : city.lng;
    const method = localStorage.getItem('noor_guest_calc_method') || 'tehran';
    const dayTimes = calculatePrayerTimes(baseCandidate, lat, lng, city.timezone, method);

    if (reminder.id === 'prayer_auto_tahajjud') {
          const npWin = calculateNightPrayerWindow(dayTimes, getNightPrayerConfig());
          const [nh, nm] = npWin.recommendedAlarmTime.split(':').map(Number);
          if (!isNaN(nh) && !isNaN(nm)) {
            const shifted = new Date(baseCandidate);
            shifted.setHours(nh, nm, 0, 0);
            return shifted;
          }
        }
        let pKey: 'fajr' | 'dhuhr' | 'asr' | 'maghrib' | 'isha' | null = null;
    if (reminder.id.includes('fajr') || reminder.title.includes('صبح')) pKey = 'fajr';
    else if (reminder.id.includes('dhuhr') || reminder.title.includes('ظهر')) pKey = 'dhuhr';
    else if (reminder.id.includes('asr') || reminder.title.includes('عصر')) pKey = 'asr';
    else if (reminder.id.includes('maghrib') || reminder.title.includes('مغرب')) pKey = 'maghrib';
    else if (reminder.id.includes('isha') || reminder.title.includes('عشاء')) pKey = 'isha';

    if (!pKey) return baseCandidate;

    let targetHHMM = dayTimes[pKey];
    if (isPreAdhan) {
      const mMatch = reminder.id.match(/pre_adhan_[a-z]+_(\d+)/);
      const offsetMin = mMatch ? Number(mMatch[1]) : 10;
      targetHHMM = addMinutesToTimeStr(targetHHMM, -offsetMin);
    }

    const [h, m] = targetHHMM.split(':').map(Number);
    if (!isNaN(h) && !isNaN(m)) {
      const shifted = new Date(baseCandidate);
      shifted.setHours(h, m, 0, 0);
      return shifted;
    }
  } catch {
    // ignore
  }
  return baseCandidate;
}

export function getUpcomingOccurrences(
  timeStr: string,
  days: number[],
  fromDate: Date = new Date(),
  horizonDays = 14
): Date[] {
  const parts = timeStr.split(':');
  if (parts.length !== 2) return [];
  const hour = Number(parts[0]);
  const minute = Number(parts[1]);
  if (isNaN(hour) || isNaN(minute) || hour < 0 || hour > 23 || minute < 0 || minute > 59) {
    return [];
  }

  const results: Date[] = [];
  for (let offset = 0; offset <= horizonDays; offset++) {
    const candidate = new Date(
      fromDate.getFullYear(),
      fromDate.getMonth(),
      fromDate.getDate() + offset,
      hour,
      minute,
      0,
      0
    );
    if (candidate <= fromDate) continue;
    if (days.length === 0 || days.includes(candidate.getDay())) {
      results.push(candidate);
    }
  }
  return results;
}

export function computeNextOccurrences(
  timeStr: string,
  days: number[],
  horizonDays = 14
): Date[] {
  return getUpcomingOccurrences(timeStr, days, new Date(), horizonDays);
}

export function computeSingleOccurrence(dateStr: string | null, timeStr: string): Date | null {
  if (!dateStr) return null;
  const [year, month, day] = dateStr.split('-').map(Number);
  const [hour, minute] = timeStr.split(':').map(Number);
  if ([year, month, day, hour, minute].some(isNaN)) return null;

  const target = new Date(year, month - 1, day, hour, minute, 0, 0);
  if (target <= new Date()) return null;
  return target;
}

const scheduledSlotMap = new Map<string, number[]>();

export async function cancelReminder(reminderId: string): Promise<void> {
  const existingIds = scheduledSlotMap.get(reminderId) ?? [];
  const fallbackIds = Array.from({ length: 15 }, (_, i) => reminderIdToNotifId(reminderId, i));
  const allIds = Array.from(new Set([...existingIds, ...fallbackIds]));

  try {
    await LocalNotifications.cancel({
      notifications: allIds.map((id) => ({ id })),
    });
  } catch {
    // ignore
  }

  if (Capacitor.getPlatform() === 'android') {
    for (const id of allIds.slice(0, 15)) {
      try {
        await NoorAlarm.cancelNativeAlarm({ id });
      } catch {
        // ignore
      }
    }
  }
  scheduledSlotMap.delete(reminderId);
}

export async function scheduleReminder(
  reminder: LocalReminder,
  globalNotifEnabled = true
): Promise<{ success: boolean; scheduledCount: number; scheduledIds: number[]; error?: string }> {
  await cancelReminder(reminder.id);

  if (!reminder.enabled || !globalNotifEnabled) {
    return { success: true, scheduledCount: 0, scheduledIds: [] };
  }

  const perm = await checkPermission();
  if (perm !== 'granted') {
    const requested = await requestPermission();
    if (requested !== 'granted') {
      return {
        success: false,
        scheduledCount: 0,
        scheduledIds: [],
        error: 'مجوز اعلان داده نشده است. لطفاً از تنظیمات گوشی، اعلان‌های برنامه نور را فعال کنید.',
      };
    }
  }

  if (typeof LocalNotifications.checkExactNotificationSetting === 'function') {
    try {
      const exactRes = await LocalNotifications.checkExactNotificationSetting();
      if (exactRes && exactRes.exact_alarm && exactRes.exact_alarm !== 'granted') {
        return {
          success: false,
          scheduledCount: 0,
          scheduledIds: [],
          error: 'برای پخش دقیق اذان و یادآوری در زمان قفل بودن صفحه، دسترسی «آلارم‌ها و یادآوری‌ها» را فعال کنید.',
        };
      }
    } catch {
      // ignore if unsupported on web
    }
  }

  await ensureChannel();
  const body = buildNotificationBody(reminder);
  const isPrayerAdhan = reminder.id.startsWith('prayer_auto_') || reminder.title.includes('اذان');
  const targetChannelId = isPrayerAdhan ? ADHAN_CHANNEL_ID : REMINDER_CHANNEL_ID;

  if (reminder.type === 'once') {
    const target = computeSingleOccurrence(reminder.date, reminder.time);
    if (!target) {
      return {
        success: false,
        scheduledCount: 0,
        scheduledIds: [],
        error: 'تاریخ و ساعت انتخاب‌شده در گذشته است. لطفاً زمان آینده انتخاب کنید.',
      };
    }

    const notifId = reminderIdToNotifId(reminder.id, 0);
    try {
      await LocalNotifications.schedule({
        notifications: [
          {
            id: notifId,
            title: reminder.title,
            body,
            channelId: targetChannelId,
            sound: Capacitor.getPlatform() === 'android' ? undefined : (isPrayerAdhan ? 'adhan_mahur' : undefined),
            schedule: { at: target, allowWhileIdle: true },
            extra: { reminderId: reminder.id, dhikrId: reminder.dhikrId ?? '' },
          },
        ],
      });
      if (Capacitor.getPlatform() === 'android') {
        try {
          const p = resolvePrayerAlarmParams(reminder.id, reminder.title);
          await NoorAlarm.scheduleNativeAlarm({
            id: notifId,
            triggerAtMillis: target.getTime(),
            title: reminder.title,
            body,
            ...p,
          });
        } catch {
          // ignore
        }
      }
      scheduledSlotMap.set(reminder.id, [notifId]);
      logEvent('info', 'ReminderService', `آلارم یک‌باره زمان‌بندی شد: ${reminder.title}`, {
        at: target.toISOString(),
      });
      return { success: true, scheduledCount: 1, scheduledIds: [notifId] };
    } catch (err) {
      logEvent('error', 'ReminderService', 'خطا در زمان‌بندی آلارم یک‌باره', { error: String(err) });
      return { success: false, scheduledCount: 0, scheduledIds: [], error: 'خطا در ثبت اعلان در سیستم‌عامل.' };
    }
  }

  const occurrences = getUpcomingOccurrences(reminder.time, reminder.days, new Date(), 14);
  if (occurrences.length === 0) {
    return { success: true, scheduledCount: 0, scheduledIds: [] };
  }

  const scheduledIds: number[] = [];
  try {
    for (let idx = 0; idx < occurrences.length; idx++) {
      const atDate = computeDynamicPrayerDate(reminder, occurrences[idx]);
      const notifId = reminderIdToNotifId(reminder.id, idx);
      scheduledIds.push(notifId);
      await LocalNotifications.schedule({
        notifications: [
          {
            id: notifId,
            title: reminder.title,
            body,
            channelId: targetChannelId,
            sound: Capacitor.getPlatform() === 'android' ? undefined : (isPrayerAdhan ? 'adhan_mahur' : undefined),
            schedule: { at: atDate, allowWhileIdle: true },
            extra: { reminderId: reminder.id, dhikrId: reminder.dhikrId ?? '' },
          },
        ],
      });
      if (Capacitor.getPlatform() === 'android') {
        try {
          const p = resolvePrayerAlarmParams(reminder.id, reminder.title);
          await NoorAlarm.scheduleNativeAlarm({
            id: notifId,
            triggerAtMillis: atDate.getTime(),
            title: reminder.title,
            body,
            ...p,
          });
        } catch {
          // ignore
        }
      }
    }
    scheduledSlotMap.set(reminder.id, scheduledIds);
    logEvent('info', 'ReminderService', `یادآور تکرارشونده زمان‌بندی شد: ${reminder.title}`, {
      count: scheduledIds.length,
      time: reminder.time,
    });
    return { success: true, scheduledCount: scheduledIds.length, scheduledIds };
  } catch (err) {
    logEvent('error', 'ReminderService', 'خطا در زمان‌بندی یادآور تکرارشونده', { error: String(err) });
    return { success: false, scheduledCount: 0, scheduledIds: [], error: 'خطا در زمان‌بندی اعلان.' };
  }
}

export async function rescheduleAllReminders(
  globalNotifEnabled = true
): Promise<{ total: number; succeeded: number; failed: string[] }> {
  const reminders = loadReminders();
  let succeeded = 0;
  const failed: string[] = [];
  for (const r of reminders) {
    const res = await scheduleReminder(r, globalNotifEnabled);
    if (res.success) {
      succeeded++;
    } else if (res.error) {
      failed.push(res.error);
    }
  }
  return { total: reminders.length, succeeded, failed };
}

export const rescheduleAll = rescheduleAllReminders;

export async function cancelAllScheduledNotifications(): Promise<void> {
  if (!Capacitor.isNativePlatform()) return;
  try {
    const pending = await LocalNotifications.getPending();
    if (pending.notifications.length > 0) {
      await LocalNotifications.cancel({
        notifications: pending.notifications.map((n) => ({ id: n.id })),
      });
    }
    scheduledSlotMap.clear();
  } catch (err) {
    console.warn('[ReminderService] cancelAll failed:', err);
  }
}

export interface ReminderMutationResult {
  reminder: LocalReminder;
  schedule: { success: boolean; error?: string };
  scheduleResult: { success: boolean; error?: string };
}

export async function createReminder(params: {
  title: string;
  body?: string;
  time: string;
  days: number[];
  dhikrId?: string | null;
  type?: 'once' | 'recurring';
  date?: string | null;
  notifEnabled?: boolean;
}): Promise<ReminderMutationResult> {
  const now = new Date().toISOString();
  const rType = params.type ?? 'recurring';
  const reminder: LocalReminder = {
    id: generateReminderId(),
    title: params.title.trim(),
    body: (params.body ?? '').trim(),
    time: params.time,
    days: rType === 'recurring' ? params.days : [],
    enabled: true,
    dhikrId: params.dhikrId ?? null,
    type: rType,
    date: rType === 'once' ? (params.date ?? null) : null,
    createdAt: now,
    updatedAt: now,
  };

  saveReminder(reminder);
  const res = await scheduleReminder(reminder, params.notifEnabled ?? true);
  const statusObj = { success: res.success, error: res.error };
  return { reminder, schedule: statusObj, scheduleResult: statusObj };
}

export async function updateReminder(
  id: string,
  patch: Partial<Omit<LocalReminder, 'id' | 'createdAt'>>,
  notifEnabled = true
): Promise<{ reminder: LocalReminder | null; schedule: { success: boolean; error?: string }; scheduleResult: { success: boolean; error?: string } }> {
  const all = loadReminders();
  const existing = all.find((r) => r.id === id);
  if (!existing) {
    const errObj = { success: false, error: 'یادآوری یافت نشد' };
    return { reminder: null, schedule: errObj, scheduleResult: errObj };
  }

  const updated: LocalReminder = {
    ...existing,
    ...patch,
    updatedAt: new Date().toISOString(),
  };

  saveReminder(updated);
  const res = await scheduleReminder(updated, notifEnabled);
  const statusObj = { success: res.success, error: res.error };
  return { reminder: updated, schedule: statusObj, scheduleResult: statusObj };
}

export async function toggleReminder(
  id: string,
  notifEnabled = true
): Promise<{ reminder: LocalReminder | null; schedule: { success: boolean; error?: string }; scheduleResult: { success: boolean; error?: string } }> {
  const all = loadReminders();
  const existing = all.find((r) => r.id === id);
  if (!existing) {
    const errObj = { success: false, error: 'یادآوری یافت نشد' };
    return { reminder: null, schedule: errObj, scheduleResult: errObj };
  }
  return updateReminder(id, { enabled: !existing.enabled }, notifEnabled);
}

export async function deleteReminder(id: string): Promise<void> {
  await cancelReminder(id);
  deleteReminderFromStorage(id);
}

export async function snoozeReminderByMinutes(title: string, body: string, minutes = 10): Promise<void> {
  const triggerAt = new Date(Date.now() + minutes * 60 * 1000);
  const snoozeId = 890000 + (Date.now() % 9000);
  if (Capacitor.getPlatform() === 'android') {
    try {
      await NoorAlarm.scheduleNativeAlarm({
        id: snoozeId,
        triggerAtMillis: triggerAt.getTime(),
        title: '💤 یادآور مجدد: ' + title,

        adhanId: 'chant_rabbanaghfir',
        durationSeconds: 15,
        volumePercent: 95,
      });
      return;
    } catch {}
  }
  try {
    await LocalNotifications.schedule({
      notifications: [
        {
          id: snoozeId,
          title: '💤 یادآور مجدد: ' + title,
          body,
          channelId: REMINDER_CHANNEL_ID,
          schedule: { at: triggerAt, allowWhileIdle: true },
        },
      ],
    });
  } catch {}
}

    private static final String PREFS_NAME = "noor_widget_prefs";

    public static void pushWidgetUpdate(Context context, String title, String nextPrayer, String subtitle) {
        try {
            SharedPreferences prefs = context.getSharedPreferences(PREFS_NAME, Context.MODE_PRIVATE);
            if (title != null && nextPrayer != null && subtitle != null) {
                prefs.edit()
                    .putString("title", title)
                    .putString("nextPrayer", nextPrayer)
                    .putString("subtitle", subtitle)
                    .apply();
            }
            String savedTitle = prefs.getString("title", "🌿 دستیار معنوی نور");
            String savedNext = prefs.getString("nextPrayer", "اوقات شرعی و یادآور اذان");
            String savedSub = prefs.getString("subtitle", "برای مشاهده اوقات شرعی و تسبیح لمس کنید");

            AppWidgetManager gm = AppWidgetManager.getInstance(context);
            ComponentName cn = new ComponentName(context, NoorPrayerWidgetProvider.class);
            int[] ids = gm.getAppWidgetIds(cn);
            if (ids != null && ids.length > 0) {
                Intent intent = new Intent(context, MainActivity.class);
                PendingIntent pendingIntent = PendingIntent.getActivity(
                    context, 0, intent, PendingIntent.FLAG_UPDATE_CURRENT | PendingIntent.FLAG_IMMUTABLE
                );
                for (int appWidgetId : ids) {
                    RemoteViews views = new RemoteViews(context.getPackageName(), R.layout.noor_prayer_widget);
                    views.setTextViewText(R.id.widget_title, savedTitle);
                    views.setTextViewText(R.id.widget_next_prayer, savedNext);
                    views.setTextViewText(R.id.widget_subtitle, savedSub);
                    views.setOnClickPendingIntent(R.id.widget_root, pendingIntent);
                    gm.updateAppWidget(appWidgetId, views);
                }
            }
        } catch (Exception ignored) {
        }
    }

    @Override
    public void onUpdate(Context context, AppWidgetManager appWidgetManager, int[] appWidgetIds) {
        pushWidgetUpdate(context, null, null, null);
    }
